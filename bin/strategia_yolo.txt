import cv2
import numpy as np

# 1. Wczytanie modelu
net = cv2.dnn.readNetFromDarknet('yolov4-tiny-custom.cfg', 'yolov4-tiny-custom_last.weights')
net.setPreferableBackend(cv2.dnn.DNN_BACKEND_CUDA) # Opcjonalne, jeśli masz GPU
net.setPreferableTarget(cv2.dnn.DNN_TARGET_CUDA)

# 2. Pobranie nazw klas
with open("obj_names.txt", "r") as f:
    classes = [line.strip() for line in f.readlines()]

cap = cv2.VideoCapture(0)

while True:
    _, frame = cap.read()
    h, w, _ = frame.shape

    # Przygotowanie obrazu dla sieci (blob)
    blob = cv2.dnn.blobFromImage(frame, 1/255, (224, 224), (0,0,0), swapRB=True, crop=False)
    net.setInput(blob)
    
    # Pobranie wyników
    output_layers = net.getUnconnectedOutLayersNames()
    layer_outputs = net.forward(output_layers)

    for output in layer_outputs:
        for detection in output:
            scores = detection[5:]
            class_id = np.argmax(scores)
            confidence = scores[class_id]

            if confidence > 0.5: # Próg pewności
                # Obliczanie współrzędnych ramki
                center_x = int(detection[0] * w)
                center_y = int(detection[1] * h)
                width = int(detection[2] * w)
                height = int(detection[3] * h)
                x = int(center_x - width / 2)
                y = int(center_y - height / 2)

                # RYSOWANIE ZIELONEGO LABELA
                cv2.rectangle(frame, (x, y), (x + width, y + height), (0, 255, 0), 2)
                label = f"{classes[class_id]}: {confidence:.2f}"
                cv2.putText(frame, label, (x, y - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 2)

    cv2.imshow("Kamera YOLO", frame)
    if cv2.waitKey(1) == 27: break # ESC zamyka

cap.release()
cv2.destroyAllWindows()
